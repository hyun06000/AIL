#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <curl/curl.h>
#include <jansson.h>
#include <unistd.h>

typedef struct {
    char *data;
    size_t size;
} response_t;

typedef struct {
    int ok;
    char *name;
    char *email;
    char *error;
} result_t;

static size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp) {
    size_t realsize = size * nmemb;
    response_t *mem = (response_t *)userp;

    char *ptr = realloc(mem->data, mem->size + realsize + 1);
    if (!ptr) {
        return 0;
    }

    mem->data = ptr;
    memcpy(&(mem->data[mem->size]), contents, realsize);
    mem->size += realsize;
    mem->data[mem->size] = 0;

    return realsize;
}

result_t fetch_json(const char *url) {
    result_t result = {0};
    result.name = malloc(512);
    result.email = malloc(512);
    result.error = malloc(1024);
    result.name[0] = '\0';
    result.email[0] = '\0';
    result.error[0] = '\0';

    CURL *curl = curl_easy_init();
    if (!curl) {
        result.ok = 0;
        strcpy(result.error, "Failed to initialize curl");
        return result;
    }

    int max_retries = 3;
    int backoff_seconds = 1;

    for (int attempt = 0; attempt < max_retries; attempt++) {
        if (attempt > 0) {
            sleep(backoff_seconds);
            backoff_seconds *= 2;
        }

        response_t response = {0};
        response.data = malloc(1);
        response.size = 0;

        curl_easy_setopt(curl, CURLOPT_URL, url);
        curl_easy_setopt(curl, CURLOPT_TIMEOUT, 5L);
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, (void *)&response);
        curl_easy_setopt(curl, CURLOPT_FAILONERROR, 1L);

        CURLcode res = curl_easy_perform(curl);

        if (res == CURLE_OK) {
            json_error_t error;
            json_t *root = json_loads(response.data, 0, &error);

            if (!root) {
                result.ok = 0;
                snprintf(result.error, 1024, "JSON parse error: %s", error.text);
                free(response.data);
                continue;
            }

            json_t *name_obj = json_object_get(root, "name");
            json_t *email_obj = json_object_get(root, "email");

            if (json_is_string(name_obj) && json_is_string(email_obj)) {
                result.ok = 1;
                strncpy(result.name, json_string_value(name_obj), 511);
                strncpy(result.email, json_string_value(email_obj), 511);
                result.name[511] = '\0';
                result.email[511] = '\0';
                result.error[0] = '\0';
                json_decref(root);
                free(response.data);
                curl_easy_cleanup(curl);
                return result;
            } else {
                result.ok = 0;
                strcpy(result.error, "Missing name or email fields in JSON");
                json_decref(root);
                free(response.data);
                continue;
            }
        } else {
            if (attempt == max_retries - 1) {
                result.ok = 0;
                snprintf(result.error, 1024, "HTTP request failed: %s", curl_easy_strerror(res));
            }
            free(response.data);
        }
    }

    curl_easy_cleanup(curl);
    return result;
}

void print_result(result_t result) {
    printf("{\n");
    printf("  \"ok\": %s", result.ok ? "true" : "false");
    if (result.ok) {
        printf(",\n  \"name\": \"%s\",\n", result.name);
        printf("  \"email\": \"%s\"\n", result.email);
    } else {
        printf(",\n  \"error\": \"%s\"\n", result.error);
    }
    printf("}\n");
}

void free_result(result_t result) {
    free(result.name);
    free(result.email);
    free(result.error);
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <url>\n", argv[0]);
        return 1;
    }

    result_t result = fetch_json(argv[1]);
    print_result(result);
    free_result(result);

    return result.ok ? 0 : 1;
}
