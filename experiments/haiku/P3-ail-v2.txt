task processJobs(jobs) {
  goal Process jobs within 10 operation budget, track results
  done budget <= 0 || processed == length(jobs)
  never []

  let budget = 10
  let done = 0
  let failed = 0
  let processed = 0

  each job in jobs {
    if budget > 0 {
      let budget = budget - 1
      let processed = processed + 1

      if job.status == "success" {
        let done = done + 1
      } else {
        let failed = failed + 1
      }
    }
  }

  let remaining = length(jobs) - processed

  return { done done, failed failed, remaining remaining }
}
