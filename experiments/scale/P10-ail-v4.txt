fn parseQuery(query) {
  let emptyCount = 0
  let result = {}

  each pair in split(query, "&") {
    let keyValue = split(pair, "=")
    let key = keyValue[0]
    let value = if len(keyValue) > 1 { keyValue[1] } else { "" }

    if value == "" {
      set emptyCount = emptyCount + 1
    }

    set result[key] = value
  }

  return { object result, count emptyCount }
}
