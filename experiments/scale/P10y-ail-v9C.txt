fn parseQueryString(queryString) {
  let pairs = split(queryString, "&")
  let result = {}
  let emptyCount = 0
  
  each pair in pairs {
    let keyValue = split(pair, "=")
    let key = first(keyValue)
    let value = if len(keyValue) > 1 { keyValue[1] } else { "" }
    
    if value == "" {
      set emptyCount = emptyCount + 1
    }
    
    set result[key] = value
  }
  
  return { object result, emptyCount emptyCount }
}
