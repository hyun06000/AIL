fn merge(list1, list2) {
  let result = []
  let i = 0
  let j = 0
  let len1 = len(list1)
  let len2 = len(list2)
  
  each _ in range(len1 + len2) {
    if i >= len1 {
      set result = push(result, list2[j])
      set j = j + 1
    } else if j >= len2 {
      set result = push(result, list1[i])
      set i = i + 1
    } else if list1[i] <= list2[j] {
      set result = push(result, list1[i])
      set i = i + 1
    } else {
      set result = push(result, list2[j])
      set j = j + 1
    }
  }
  
  return { merged result, length len(result) }
}
