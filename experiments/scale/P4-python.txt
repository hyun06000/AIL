def merge_sorted_lists(list1, list2):
    """
    Merge two sorted lists into one sorted list without using built-in sort.
    Returns the merged list and its length.
    """
    merged = []
    i = 0  # pointer for list1
    j = 0  # pointer for list2

    # Merge elements from both lists
    while i < len(list1) and j < len(list2):
        if list1[i] <= list2[j]:
            merged.append(list1[i])
            i += 1
        else:
            merged.append(list2[j])
            j += 1

    # Append remaining elements from list1
    while i < len(list1):
        merged.append(list1[i])
        i += 1

    # Append remaining elements from list2
    while j < len(list2):
        merged.append(list2[j])
        j += 1

    return merged, len(merged)


# Test the function
if __name__ == "__main__":
    list1 = [1, 3, 5, 7]
    list2 = [2, 4, 6, 8]

    merged_list, length = merge_sorted_lists(list1, list2)
    print(f"Merged list: {merged_list}")
    print(f"Length: {length}")
